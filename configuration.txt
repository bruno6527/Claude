    actions:
      - variables:
          # Capteurs qui doivent envoyer une valeur régulièrement

            - sensor.living_room_temperature_salon_air_temperature

            - sensor.living_room_tablette_esp_page_ecran
            - sensor.living_room_temperature_salon_node_status
            - sensor.bedroom_sensor_chambre_node_status
            - sensor.node_5_node_status
          muets: >-
            {%- set delai = states('input_number.delai_capteur_muet') | float(3) * 3600 -%}
            {%- set ns = namespace(l=[]) -%}
            {%- for e in surveilles -%}
              {%- set s = states[e] -%}
              {%- if s is none -%}
                {%- set ns.l = ns.l + [{'nom': e, 'info': 'introuvable'}] -%}
              {%- elif s.state in ['unavailable', 'unknown'] -%}
                {%- set ns.l = ns.l + [{'nom': s.name, 'info': 'indisponible'}] -%}
              {%- else -%}
                {%- set age = (now() - s.last_reported).total_seconds() -%}
                {%- if age > delai -%}
                  {%- set ns.l = ns.l + [{'nom': s.name, 'info': 'muet depuis ' ~ (age / 3600) | round(1) ~ ' h'}] -%}
                {%- endif -%}
              {%- endif -%}
            {%- endfor -%}
            {%- for e in joignables -%}
              {%- set s = states[e] -%}
              {%- if s is none or s.state in ['unavailable', 'unknown', 'dead'] -%}

              {%- endif -%}
            {%- endfor -%}
